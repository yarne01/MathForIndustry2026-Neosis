\section{Metric Distance Definition}
\begin{frame}{Metric Distance}
    \begin{enumerate}
        \item Why it is needed
        \item What had been done in the project
        \item What was expanded upon, which ones were introduced
        \begin{enumerate}
            \item Hausdorff
            \item Chamfer
            \item Wasserstein
            \item Custom SDF-INT solution 
        \end{enumerate}
        \item What objects are we measuring with this metric function?
    \end{enumerate}

    $$ d(\vec{x}, Y) = \inf_{\vec{y} \in Y} d(\vec{x}, \vec{y}) $$
\end{frame}

\subsection{Hausdorff}
\begin{frame}{Hausdorff}

    \begin{columns}
        \begin{column}{0.6\textwidth}

            Traditional definition:
            
            $$ d_H(X,Y) := \max \left\{ \sup_{\vec{x} \in X}  d(\vec{x}, Y)  , \sup_{\vec{y} \in Y}  d(X, \vec{y})  \right\} $$

            Definition within the existing library: 

            $$ \tilde{d}_H(X,Y) = \sup_{\vec{x} \in X} \left( d(\vec{x}, Y) ^2\right) + \sup_{\vec{y} \in Y} \left( d(X, \vec{y}) ^2\right) $$
        \end{column}
        
        \begin{column}{0.3\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\linewidth]{figs/Hausdorff_distance_sample.png}
                \caption{$\sup \inf$ visualisation \tiny{(\url{https://en.wikipedia.org/wiki/Hausdorff_distance})}}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}






\subsection{Chamfer}
\begin{frame}{Chamfer}

    \begin{columns}
        \begin{column}{0.6\textwidth}

            \begin{enumerate}
                \item Not True metric, fails $\triangle$ inequality

                $$ d_C(X,Y) = \sum_{\vec{x} \in X} d(\vec{x}, Y)  + \sum_{\vec{y} \in Y} d(X, \vec{y}) $$

                
                \item Same issue, but different :D

                $$ \tilde{d}_C(X,Y) = \Imp[mean]_{\vec{x} \in X} \left( d(\vec{x}, Y)  \right) + \Imp[mean]_{\vec{y} \in Y} \left( d(X, \vec{y}) ) \right) $$
                
            \end{enumerate}
            
            
        \end{column}
        \begin{column}{0.3\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\linewidth]{figs/Hausdorff_distance_sample.png}
                \caption{$\sup \inf$ visualisation \tiny{(\url{https://en.wikipedia.org/wiki/Hausdorff_distance})}}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}



\section{Custom SDF-INT measure}
\begin{frame}{Custom SDF-INT measure}
    $$ d_I := \int_{\Omega} \left\| d(\vec{\omega}, X), d(\vec{\omega}, Y) \right\| d\vec{\omega}$$

    $$\Omega = [x_1, x_2] \times [y_1, y_2] \times [z_1, z_2]$$

    
\end{frame}
